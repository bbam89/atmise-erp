// ===== Atmise ERP - Google Apps Script ===== const SHEET_ID = '1JmIFU82kAnKJ0ITOdzrI5pmqpTrH5ksbBgBBZQ9HsyE'; const SHEETS = { orders: '주문', stocks: '재고', settles: '정산', purchases: '발주', losses: '로스', channels: '채널', }; const HEADERS = { orders: ['id','date','channel','orderNum','buyer','productName','sku','qty','price','settle','ship','track','status','note','settled','settleDate'], stocks: ['sku','name','line','qty','inQty','outQty','lossQty','avgPrice','safe','note'], settles: ['id','month','channel','sales','fee','ad','etc','net','note'], purchases: ['id','date','vendor','sku','qty','price','due','status','note'], losses: ['id','date','sku','qty','type','deduct','note'], channels: ['id','name','fee','cycle','contact','note'], }; function setupSheets() { const ss = SpreadsheetApp.openById(SHEET_ID); Object.entries(SHEETS).forEach(([key, sheetName]) => { let sheet = ss.getSheetByName(sheetName); if (!sheet) sheet = ss.insertSheet(sheetName); if (sheet.getLastRow() === 0) { sheet.appendRow(HEADERS[key]); sheet.getRange(1, 1, 1, HEADERS[key].length) .setBackground('#1D9E75') .setFontColor('#ffffff') .setFontWeight('bold'); sheet.setFrozenRows(1); } }); return '시트 세팅 완료'; } function doGet(e) { const action = e.parameter.action; const sheet_key = e.parameter.sheet; try { if (action === 'setup') return jsonResponse({ok: true, message: setupSheets()}); if (action === 'read') return jsonResponse({ok: true, data: readSheet(sheet_key)}); if (action === 'readAll') { const all = {}; Object.keys(SHEETS).forEach(key => { all[key] = readSheet(key); }); return jsonResponse({ok: true, data: all}); } return jsonResponse({ok: false, message: '알 수 없는 action'}); } catch(err) { return jsonResponse({ok: false, message: err.toString()}); } } function doPost(e) { try { const payload = JSON.parse(e.postData.contents); const action = payload.action; const sheet_key = payload.sheet; const data = payload.data; if (action === 'saveAll') { Object.entries(data).forEach(([key, rows]) => { if (SHEETS[key]) writeSheet(key, rows); }); return jsonResponse({ok: true, message: '전체 저장 완료'}); } if (action === 'write') { writeSheet(sheet_key, data); return jsonResponse({ok: true, message: `${sheet_key} 저장 완료`}); } return jsonResponse({ok: false, message: '알 수 없는 action'}); } catch(err) { return jsonResponse({ok: false, message: err.toString()}); } } function readSheet(key) { const ss = SpreadsheetApp.openById(SHEET_ID); const sheet = ss.getSheetByName(SHEETS[key]); if (!sheet || sheet.getLastRow() <= 1) return []; const headers = HEADERS[key]; const rows = sheet.getRange(2, 1, sheet.getLastRow() - 1, headers.length).getValues(); return rows .filter(row => row[0] !== '' && row[0] !== null) .map(row => { const obj = {}; headers.forEach((h, i) => { obj[h] = row[i]; }); return obj; }); } function writeSheet(key, rows) { const ss = SpreadsheetApp.openById(SHEET_ID); const sheetName = SHEETS[key]; let sheet = ss.getSheetByName(sheetName); if (!sheet) sheet = ss.insertSheet(sheetName); const headers = HEADERS[key]; // 주문 시트는 헤더 컬럼 수 자동 맞춤 const currentHeaders = sheet.getLastRow() > 0 ? sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0] : []; if (currentHeaders.length < headers.length) { sheet.getRange(1, 1, 1, headers.length).setValues([headers]); sheet.getRange(1, 1, 1, headers.length) .setBackground('#1D9E75') .setFontColor('#ffffff') .setFontWeight('bold'); sheet.setFrozenRows(1); } if (sheet.getLastRow() > 1) sheet.deleteRows(2, sheet.getLastRow() - 1); if (sheet.getLastRow() === 0) { sheet.appendRow(headers); sheet.getRange(1, 1, 1, headers.length) .setBackground('#1D9E75') .setFontColor('#ffffff') .setFontWeight('bold'); sheet.setFrozenRows(1); } if (rows && rows.length > 0) { const values = rows.map(row => headers.map(h => { const v = row[h]; return (v === undefined || v === null) ? '' : v; })); sheet.getRange(2, 1, values.length, headers.length).setValues(values); } } function jsonResponse(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
